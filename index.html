ctx.moveTo(snake[0].x, snake[0].y); for (let i = 1; i < snake.length; i++) { const previous = snake[i - 1]; const current = snake[i]; const dx = Math.abs(current.x - previous.x); const dy = Math.abs(current.y - previous.y); /* اگر مار از دیوار رد شده باشد، فاصله بین دو نقطه خیلی زیاد می‌شود. بنابراین خط را قطع می‌کنیم. */ if (dx > GRID * 1.5 || dy > GRID * 1.5) { ctx.moveTo( current.x, current.y ); } else { ctx.lineTo( current.x, current.y ); } } ctx.strokeStyle = "#4dff00"; ctx.lineWidth = 30; ctx.shadowColor = "#52ff00"; ctx.shadowBlur = 22; ctx.stroke(); /* خط روشن وسط بدن */ ctx.beginPath(); ctx.moveTo( snake[0].x, snake[0].y - 3 ); for (let i = 1; i < snake.length; i++) { const previous = snake[i - 1]; const current = snake[i]; const dx = Math.abs(current.x - previous.x); const dy = Math.abs(current.y - previous.y); if (dx > GRID * 1.5 || dy > GRID * 1.5) { ctx.moveTo( current.x, current.y - 3 ); } else { ctx.lineTo( current.x, current.y - 3 ); } } ctx.strokeStyle = "#b4ff50"; ctx.lineWidth = 8; ctx.shadowBlur = 5; ctx.stroke(); /* ========================== سر مار ========================== */ const h = snake[0]; const sx = -direction.y * 8; const sy = direction.x * 8; const fx = direction.x * 8; const fy = direction.y * 8; const eye1 = { x: h.x + fx + sx, y: h.y + fy + sy }; const eye2 = { x: h.x + fx - sx, y: h.y + fy - sy }; ctx.stroke(); ctx.restore(); }
